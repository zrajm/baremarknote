   Test for uniqueness, so as to not overwrite any note.
   ----------------------------------------
   LC_ALL="C"

   # Create 11-char URL-safe base64 string, dash-builtins only.
   random_id() {
       local ID CHUNK CHR
       while [ "${#ID}" -lt 11 ]; do
           IFS='' read -r CHUNK </dev/urandom
           while [ -n "$CHUNK" ] && [ "${#ID}" -lt 11 ]; do
               CHR="${CHUNK%"${CHUNK#?}"}"
               CHUNK="${CHUNK#?}"
               case "$CHR" in [a-zA-Z0-9_-]) ID="$ID$CHR"; esac
           done
       done
       printf '%s\n' "$ID"
   }
